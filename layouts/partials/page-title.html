{{- /* The page's own title, without the site's. An untitled post takes its first sentence, cut at a word boundary near 60 characters. Returns "" on the homepage, or when an untitled post has no text. */ -}} {{- $title := "" -}} {{- if .IsHome -}} {{- else if .Title -}} {{- $title = .Title -}} {{- else -}} {{- $text := .Plain | replaceRE `\s+` " " | strings.TrimSpace -}} {{- with findRE `^.*?[.!?](\s|$)` $text 1 }}{{ $text = index . 0 | strings.TrimSpace }}{{ end -}} {{- $title = $text | truncate 60 -}} {{- end -}} {{- return $title -}}
