{{- /* The page's description: the site's on the homepage, otherwise the post's Summary field if set, else its text cut at a word boundary under 155 characters. Decoded to plain text, since both can hold HTML entities. Returns "" when there is none. */ -}} {{- $description := "" -}} {{- if .IsHome -}} {{- $description = .Site.Params.itunes_description | default "" | htmlUnescape -}} {{- else if .Params.summary -}} {{- $description = .Params.summary | htmlUnescape -}} {{- else -}} {{- $description = .Summary | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | truncate 152 -}} {{- end -}} {{- return $description -}}
